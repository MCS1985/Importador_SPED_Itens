            WHEN registro = 'C170' THEN GREATEST((vl_liquido - vl_icms), 0)

    aliq_pis AS "Alíq. Pis do SPED",
    aliq_cofins AS "Alíq. Cofins do SPED",
    nat_bc_cred_final AS "Nat. BC. Créd.",
    st_conf AS "status_conferencia",
    COUNT(*) AS "qtd_registros",
    COALESCE(SUM(vl_item), 0)::numeric(18,2) AS "total_item",
    COALESCE(SUM(vl_desc), 0)::numeric(18,2) AS "total_desconto",
    COALESCE(SUM(vl_liquido), 0)::numeric(18,2) AS "total_liquido",
    COALESCE(SUM(vl_bc_icms), 0)::numeric(18,2) AS "total_bc_icms",
    COALESCE(SUM(vl_icms), 0)::numeric(18,2) AS "total_icms",
    COALESCE(SUM(vl_icms_st), 0)::numeric(18,2) AS "total_icms_st",
    COALESCE(SUM(vl_ipi), 0)::numeric(18,2) AS "total_ipi",
    COALESCE(SUM(vl_iss), 0)::numeric(18,2) AS "total_iss",
    COALESCE(SUM(bc_pis_cof_calculada_base), 0)::numeric(18,2) AS "bc_pis_cof_calculada",
    COALESCE(SUM(bc_pis_cof_escriturada), 0)::numeric(18,2) AS "bc_pis_cof_escriturada",
    COALESCE(SUM(v_pis), 0)::numeric(18,2) AS "total_pis",
    COALESCE(SUM(v_cof), 0)::numeric(18,2) AS "total_cof",
    COALESCE(SUM(v_pis + v_cof), 0)::numeric(18,2) AS "total_pis_cofins"
FROM (
    SELECT
        *,
        -- Tratamento centralizado da Natureza da BC
        CASE
            WHEN sentido = 'Saída' THEN 'N/A'
            WHEN nat_bc_cred IS NULL
              OR TRIM(nat_bc_cred::text) = ''
              OR TRIM(nat_bc_cred::text) IN ('0', '0,0', '0,00', '0.00')
                THEN 'N/A'
            ELSE nat_bc_cred::text
        END AS nat_bc_cred_final,
        
        -- Tratamento centralizado do CST (Blindagem Nível Máximo)
        CASE
            WHEN cst IS NULL THEN '-'
            WHEN REGEXP_REPLACE(cst::text, '[^0-9]', '', 'g') IN ('', '0', '00', '000', '0000')
                THEN '-'
            ELSE TRIM(cst::text)
        END AS cst_final

    FROM BASE
) b
GROUP BY 
    SUBSTRING(dt_ref FROM 5 FOR 4),
    SUBSTRING(dt_ref FROM 3 FOR 2),
    periodo_competencia,
    ano_competencia,
    mes_competencia,
    sped_origem,
    sentido,
    registro,
    cfop,
    cst_final,
    aliq_pis,
    aliq_cofins,
    nat_bc_cred_final,
    st_conf
ORDER BY 
    "Ano" ASC, "Mês" ASC, "Registro" ASC;
